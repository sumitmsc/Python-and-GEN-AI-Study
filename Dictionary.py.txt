# ==========================================
# Basic Dictionary Example
# ==========================================

# Creating a dictionary
student_info = {
    "Name": "Sumit",
    "Email": "g@mail.com",
    "Phone": 1232
}

# Printing complete dictionary
print(student_info)

# OUTPUT:
# {'Name': 'Sumit', 'Email': 'g@mail.com', 'Phone': 1232}


# Print all keys
print(student_info.keys())

# OUTPUT:
# dict_keys(['Name', 'Email', 'Phone'])


# Print all values
print(student_info.values())

# OUTPUT:
# dict_values(['Sumit', 'g@mail.com', 1232])


# Print all key-value pairs
print(student_info.items())

# OUTPUT:
# dict_items([('Name', 'Sumit'),
#             ('Email', 'g@mail.com'),
#             ('Phone', 1232)])


# Access value using get()
print(student_info.get('Phone'))

# OUTPUT:
# 1232


# ==========================================
# Dictionary Copy Example
# ==========================================

print("\n===== Dictionary Copy Example =====")

dict_original = {
    "Name": "Amit",
    "a": 23,
    "b": 56,
    "c": 23
}

# Reference copy
dict_reference = dict_original

# Shallow copy
dict_copied = dict_original.copy()

print(dict_reference)

# OUTPUT:
# {'Name': 'Amit', 'a': 23, 'b': 56, 'c': 23}


# Updating original dictionary
dict_original['a'] = 46

print(dict_reference)

# OUTPUT:
# {'Name': 'Amit', 'a': 46, 'b': 56, 'c': 23}
#
# Explanation:
# dict_reference also changes because it points
# to the same dictionary object.


print(dict_original)

# OUTPUT:
# {'Name': 'Amit', 'a': 46, 'b': 56, 'c': 23}


print(dict_copied)

# OUTPUT:
# {'Name': 'Amit', 'a': 23, 'b': 56, 'c': 23}
#
# Explanation:
# dict_copied remains unchanged because copy()
# creates a separate dictionary object.


# ==========================================
# Nested Dictionary Example
# ==========================================

print("\n===== Nested Dictionary Example =====")

nested_dict = {

    "Student1": {
        "Name": "Amit",
        "Marks": 84,
        "Grade": "A"
    },

    "Student2": {
        "Name": "Vivek",
        "Marks": 93,
        "Grade": "A+"
    },

    "Student3": {
        "Name": "Vikas",
        "Marks": 98,
        "Grade": "A+"
    }
}

print(nested_dict)

# OUTPUT:
# {
#   'Student1': {'Name': 'Amit', 'Marks': 84, 'Grade': 'A'},
#   'Student2': {'Name': 'Vivek', 'Marks': 93, 'Grade': 'A+'},
#   'Student3': {'Name': 'Vikas', 'Marks': 98, 'Grade': 'A+'}
# }


# ==========================================
# Iterating Nested Dictionary
# ==========================================

print("\n===== Iterating Nested Dictionary =====")

for outer_key, outer_value in nested_dict.items():

    print(f"\nOuter Key : {outer_key}")
    print(f"Outer Value : {outer_value}")

    # Iterating inner dictionary
    for inner_key, inner_value in outer_value.items():
        print(f"{inner_key} : {inner_value}")


# SAMPLE OUTPUT:
#
# Outer Key : Student1
# Outer Value : {'Name': 'Amit', 'Marks': 84, 'Grade': 'A'}
# Name : Amit
# Marks : 84
# Grade : A


# ==========================================
# Dictionary Comprehension
# ==========================================

print("\n===== Dictionary Comprehension =====")

# Creating dictionary using comprehension
square_dict = {x: x**2 for x in range(11)}

print(square_dict)

# OUTPUT:
# {0: 0, 1: 1, 2: 4, 3: 9, 4: 16,
#  5: 25, 6: 36, 7: 49, 8: 64,
#  9: 81, 10: 100}


# ==========================================
# Even Number Dictionary
# ==========================================

print("\n===== Even Number Dictionary =====")

# Dictionary with only even numbers
even_square_dict = {
    x: x**2 for x in range(1, 11)
    if x % 2 == 0
}

print(even_square_dict)

# OUTPUT:
# {2: 4, 4: 16, 6: 36, 8: 64, 10: 100}


# ==========================================
# Merging Dictionaries
# ==========================================

print("\n===== Merging Dictionaries =====")

dict1 = {'a': 14, 'b': 2, 'c': 3}
dict2 = {'a': 1, 'b': 4, 'c': 7}

# Merge dictionaries using **
merged_dict = {**dict1, **dict2}

print(merged_dict)

# OUTPUT:
# {'a': 1, 'b': 4, 'c': 7}
#
# Explanation:
# If same keys exist, values from second
# dictionary overwrite first dictionary values.
